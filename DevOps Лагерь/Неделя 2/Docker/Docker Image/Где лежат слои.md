В Docker (overlay2):
/var/lib/docker/
└── overlay2/
    └── $layer-id/
        ├── diff/       # файловая система слоя
        ├── lower       # список родительских слоёв
        ├── merged/     # overlay mount (появляется только когда создаётся контейнер сам и есть только в rw снапшоте)
        └── work/       # служебная директория

/var/lib/containerd/
└── io.containerd.content.v1.content/  # blobs (слои и config)
└── io.containerd.snapshotter.v1.overlayfs/
    └── snapshots/                     # unpack слоёв
