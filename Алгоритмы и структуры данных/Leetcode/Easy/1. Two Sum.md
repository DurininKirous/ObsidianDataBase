```
func twoSum(nums []int, target int) []int {
	numsMap := make(map[int]int)
	for i, num := range nums {
		diff := target - num
		_, ok := numsMap[diff]
		if ok && numsMap[diff] != i {
			return []int{i, numsMap[diff]}
		}
		numsMap[num] = i
	}
	
	return []int{0,1}
}
```

Работает за O(n)
Цикл проходит по массиву, добавляет элементы в хеш таблицу. На каждой итерации ищет нужную разницу элементов, смотрит в хеш таблицу, делает это за O(1), в итоге сложность O(n)
