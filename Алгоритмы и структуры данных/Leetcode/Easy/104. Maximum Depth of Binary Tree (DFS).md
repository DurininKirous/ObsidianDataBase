```go
type NodeDepth struct {
    node  *TreeNode
    depth int
}

func maxDepth(root *TreeNode) int {
    if root == nil {
        return 0
    }

    queue := []NodeDepth{{root, 1}}
    maxVal := 1

    for len(queue) > 0 {
        cur := queue[0]
        queue = queue[1:]

        if cur.depth > maxVal {
            maxVal = cur.depth
        }

        if cur.node.Left != nil {
            queue = append(queue, NodeDepth{cur.node.Left, cur.depth + 1})
        }
        if cur.node.Right != nil {
            queue = append(queue, NodeDepth{cur.node.Right, cur.depth + 1})
        }
    }

    return maxVal
}
```
## Концепция
- Глубина дерева = **максимум глубины левого и правого поддерева + 1**.
- Можно решить:
    1. **DFS (рекурсивно)** → самое короткое и популярное собеседовательное решение.
    2. **BFS (через очередь)** → идём по уровням.
        - можно хранить глубину в мапе,
        - можно хранить `(узел, глубина)` прямо в очереди (эффективнее).
