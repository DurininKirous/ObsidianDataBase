```go
func minDepth(root *TreeNode) int {
    type NodeLevel struct {
        Node  *TreeNode
        Level int
    }
    if root == nil {
        return 0
    }
    queue := []*NodeLevel{{Node: root, Level: 0}}

    for len(queue) > 0 {
        node := queue[0]
        queue = queue[1:]

        // если лист → возвращаем уровень+1
        if node.Node.Left == nil && node.Node.Right == nil {
            return node.Level + 1
        }

        if node.Node.Left != nil {
            queue = append(queue, &NodeLevel{Node: node.Node.Left, Level: node.Level + 1})
        }
        if node.Node.Right != nil {
            queue = append(queue, &NodeLevel{Node: node.Node.Right, Level: node.Level + 1})
        }
    }
    return 0
}
```

## Идеи решения
### 1. BFS (оптимальный)
- Проходим дерево по уровням.
- Как только встречаем **лист** → сразу возвращаем глубину.
- Это срабатывает, потому что BFS идёт «в ширину» и первым находит кратчайший путь.
