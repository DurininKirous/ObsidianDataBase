```go
func levelOrder(root *TreeNode) [][]int {
    type NodeLevel struct {
        Node  *TreeNode
        Level int
    }
    res := [][]int{}
    if root == nil {
        return res
    }

    queue := []*NodeLevel{{Node: root, Level: 0}}

    for len(queue) > 0 {
        nodeL := queue[0]
        queue = queue[1:]

        if len(res) < nodeL.Level+1 {
            res = append(res, []int{})
        }
        res[nodeL.Level] = append(res[nodeL.Level], nodeL.Node.Val)

        if nodeL.Node.Left != nil {
            queue = append(queue, &NodeLevel{Node: nodeL.Node.Left, Level: nodeL.Level + 1})
        }
        if nodeL.Node.Right != nil {
            queue = append(queue, &NodeLevel{Node: nodeL.Node.Right, Level: nodeL.Level + 1})
        }
    }
    return res
}
```
## Концепция
- Используем **BFS** (обход в ширину).
- В очередь кладём узлы, достаём их по уровням.
- Каждый уровень формируем в отдельный список.
