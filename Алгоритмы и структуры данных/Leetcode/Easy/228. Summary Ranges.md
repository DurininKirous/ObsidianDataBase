```go
import "strconv"

func summaryRanges(nums []int) []string {
    n := len(nums)
    res := make([]string, 0, n)
    if n == 0 {
        return res
    }

    start := nums[0]

    flush := func(a, b int) {
        if a == b {
            res = append(res, strconv.Itoa(a))
        } else {
            res = append(res, strconv.Itoa(a)+"->"+strconv.Itoa(b))
        }
    }

    for i := 1; i < n; i++ {
        if nums[i] != nums[i-1]+1 {
            flush(start, nums[i-1])
            start = nums[i]
        }
    }
    flush(start, nums[n-1])
    return res
}
```
**Идея**  
Идём одним проходом. Держим начало текущего диапазона `start`.  
Когда последовательность прерывается (текущий `nums[i] != nums[i-1] + 1`), «сбрасываем» диапазон в ответ (одиночное число или `"a->b"`), затем начинаем новый диапазон с `nums[i]`. В конце не забудь финальный «сброс».

**Сложность**: `O(n)` времени, `O(1)` доп.памяти (кроме ответа).
