/**
 * Definition for singly-linked list.
 * type ListNode struct {
 *     Val int
 *     Next *ListNode
 * }
 */
func mergeTwoLists(list1 *ListNode, list2 *ListNode) *ListNode {
    res := &ListNode{}
    dummy := res
    for (list1 != nil || list2 != nil) {
        dummy.Next = &ListNode{}
        dummy = dummy.Next
        if list1 == nil {
            dummy.Val = list2.Val
            list2 = list2.Next
        } else if list2 == nil {
            dummy.Val = list1.Val
            list1 = list1.Next
        } else {
            if (list1.Val <= list2.Val) {
                dummy.Val = list1.Val
                list1 = list1.Next
            } else {
                dummy.Val = list2.Val
                list2 = list2.Next
            }
        }
    }
    return res.Next
}

## Алгоритм
- Создать фиктивный узел `res` и указатель `dummy` для построения списка.
- Пока хотя бы один список не пуст:
    - Если один список пуст → брать элементы из другого.
    - Если оба не пусты → выбрать меньший `Val`, двигать указатель соответствующего списка.
    - Создать новый узел и присоединить к результату.
- Вернуть `res.Next`, т.к. `res` — фиктивный узел.
