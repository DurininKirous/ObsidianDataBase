```python
class MaxStack:
    def __init__(self):
        self.stack = []
        self.max = []

    def push(self, x):
        self.stack.append(x)
        if not self.max:
            self.max.append(x)
        else:
            self.max.append(max(x, self.max[-1]))

    def pop(self):
        res = None
        if self.stack:
            res = self.stack.pop()
            self.max.pop()
        return res

    def top(self):
        res = None
        if self.stack:
            res = self.stack[-1]
        return res

    def peekMax(self):
        res = None
        if self.max:
            res = self.max[-1]
        return res

    def popMax(self):
        if not self.stack:
            return None

        current_max = self.peekMax()
        buffer = []
        while self.top() != current_max:
            buffer.append(self.pop())
        res = self.pop()
        while buffer:
            self.push(buffer.pop())

        return res
```
**Идея:**  
Два стека: `stack` хранит значения, `max` хранит максимум на каждом уровне. `push` кладёт значение и новый максимум. `popMax` временно вынимает элементы, пока не дойдём до текущего максимума, удаляет его, затем возвращает элементы обратно через `push`, чтобы восстановить max-стек.
