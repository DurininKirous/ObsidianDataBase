```go
// type Node struct {
//     Val       int
//     Neighbors []*Node
// }

func cloneGraph(node *Node) *Node {
    if node == nil { return nil }

    // Фаза 1: обойти граф и создать все вершины-копии (без соседей)
    clones := map[*Node]*Node{}
    q := []*Node{node}
    clones[node] = &Node{Val: node.Val}

    for len(q) > 0 {
        u := q[0]; q = q[1:]
        for _, v := range u.Neighbors {
            if _, ok := clones[v]; !ok {
                clones[v] = &Node{Val: v.Val}
                q = append(q, v)
            }
        }
    }

    // Фаза 2: проставить рёбра (neighbors) между уже созданными копиями
    for orig, copy := range clones {
        for _, v := range orig.Neighbors {
            copy.Neighbors = append(copy.Neighbors, clones[v])
        }
    }

    return clones[node]
}
```
## Идея
- **Фаза 1:** один раз обходим граф, создаём пустые копии всех узлов и кладём `orig -> copy` в `clones`.
- **Фаза 2:** пробегаем по `clones` и заполняем `Neighbors` у копий, используя уже готовые `clones[v]`.

## Сложность
- Время: `O(V + E)`
- Память: `O(V)`
