```go
func reverseVowels(s string) string {
    if len(s) < 2 { return s }

    vowels := map[byte]struct{}{
        'a': {}, 'e': {}, 'i': {}, 'o': {}, 'u': {},
        'A': {}, 'E': {}, 'I': {}, 'O': {}, 'U': {},
    }
    sb := []byte(s)
    left, right := 0, len(sb)-1

    isVowel := func(b byte) bool {
        _, ok := vowels[b]
        return ok
    }

    for left < right {
        for left < right && !isVowel(sb[left]) { left++ }
        for left < right && !isVowel(sb[right]) { right-- }
        if left < right {
            sb[left], sb[right] = sb[right], sb[left]
            left++
            right--
        }
    }
    return string(sb)
}
```
## Идея
- Используем два указателя (`left`/`right`), двигаем их к центру, пропуская негласные.
- Когда оба указывают на гласные — меняем местами и продолжаем.
- Работаем с `[]byte`, потому что `string` в Go неизменяем.
