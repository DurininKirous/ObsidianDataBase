```go
func isSymmetric(root *TreeNode) bool {
    if root == nil {
        return true
    }

    // очередь пар узлов
    queue := [][2]*TreeNode{{root.Left, root.Right}}

    for len(queue) > 0 {
        pair := queue[0]
        queue = queue[1:]
        left, right := pair[0], pair[1]

        if left == nil && right == nil {
            continue
        }
        if left == nil || right == nil || left.Val != right.Val {
            return false
        }

        // кладём зеркальные пары
        queue = append(queue, [2]*TreeNode{left.Left, right.Right})
        queue = append(queue, [2]*TreeNode{left.Right, right.Left})
    }

    return true
}

```
## Концепция
- Начинаем с пары `(root.Left, root.Right)`.
- В цикле достаём пары и проверяем:
    - оба `nil` → ок,
    - один `nil` или разные значения → `false`.
- Добавляем в очередь **зеркальные пары**:
    - `(left.Left, right.Right)`
    - `(left.Right, right.Left)`
