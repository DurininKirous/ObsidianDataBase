```go
/**

* Definition for a binary tree node.

* type TreeNode struct {

* Val int

* Left *TreeNode

* Right *TreeNode

* }

*/

func maxDepth(root *TreeNode) int {

if root == nil {

return 0

}

  

type NodeDepth struct {

node *TreeNode

depth int

}

  

queue := []NodeDepth{{root, 1}}

maxVal := 1

  

for len(queue) > 0 {

cur := queue[0]

queue = queue[1:]

  

if cur.depth > maxVal {

maxVal = cur.depth

}

  

if cur.node.Left != nil {

queue = append(queue, NodeDepth{cur.node.Left, cur.depth + 1})

}

if cur.node.Right != nil {

queue = append(queue, NodeDepth{cur.node.Right, cur.depth + 1})

}

}

  

return maxVal

}

```
## Концепция
- Начинаем с пары `(root.Left, root.Right)`.
- В цикле достаём пары и проверяем:
    - оба `nil` → ок,
    - один `nil` или разные значения → `false`.
- Добавляем в очередь **зеркальные пары**:
    - `(left.Left, right.Right)`
    - `(left.Right, right.Left)`
