```go
import "strconv"
import "strings"

func findPaths(root *TreeNode, curr []int, res *[]string) {
    if root == nil {
        return
    }

    curr = append(curr, root.Val)

    if root.Left == nil && root.Right == nil {
        path := make([]string, len(curr))
        for i, val := range curr {
            path[i] = strconv.Itoa(val)
        }
        *res = append(*res, strings.Join(path, "->"))
        return
    }
    
    findPaths(root.Left, curr, res)
    findPaths(root.Right, curr, res)
}

func binaryTreePaths(root *TreeNode) []string {
    if root == nil {
        return []string{}
    }
    res := []string{}
    findPaths(root, []int{}, &res)
    return res
}
```

## Идея
- Используем **DFS**.
- Держим путь как `[]int`.
- Когда доходим до листа → преобразуем числа в строку и сохраняем в результат.
- Преобразование в строку выполняем **только один раз в листе**.
