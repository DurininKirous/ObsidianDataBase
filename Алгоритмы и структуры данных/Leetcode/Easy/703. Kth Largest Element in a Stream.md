```go
import "container/heap"

type MinHeap []int

func (h MinHeap) Len() int           { return len(h) }
func (h MinHeap) Less(i, j int) bool { return h[i] < h[j] } 
func (h MinHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *MinHeap) Push(x interface{}) { *h = append(*h, x.(int)) }
func (h *MinHeap) Pop() interface{} {
    old := *h
    n := len(old)
    x := old[n-1]
    *h = old[:n-1]
    return x
}

type KthLargest struct {
    Number int 
    heap   *MinHeap 
}

func Constructor(k int, nums []int) KthLargest {
    h := &MinHeap{}
    heap.Init(h)

    kth := KthLargest{Number: k, heap: h}
    for _, num := range nums {
        heap.Push(kth.heap, num)
        if kth.heap.Len() > k {
            heap.Pop(kth.heap)
        }
    }
    return kth
}

func (this *KthLargest) Add(val int) int {
    heap.Push(this.heap, val)
    if this.heap.Len() > this.Number {
        heap.Pop(this.heap)
    }
    return (*this.heap)[0]
}

```

### Основная идея
Тебе дан поток чисел и число `k`.  
Нужно в любой момент уметь быстро возвращать **k-й по величине элемент** (не сортируя всё заново).

👉 Для этого идеально подходит **min-heap** размером `k`:
- Куча хранит только `k` самых больших элементов.
- Минимальный элемент в куче — это **k-й по величине**.

---

### ⚙️ Что происходит
1. В конструкторе (`Constructor`) создаётся пустая куча.
2. В неё добавляются все исходные элементы.  
    Если элементов больше `k`, удаляется минимум (`heap.Pop`).
3. Каждый новый `Add(val)`:
    - Добавляет элемент в кучу.
    - Если куча > `k` — удаляет минимум.
    - Возвращает вершину кучи (`(*h)[0]`) — это `k`-й по величине элемент.### Основная идея
