```go
func findMax(root *TreeNode, maxVal *int) int {
    if root == nil {
        return 0
    }

    left := findMax(root.Left, maxVal)
    right := findMax(root.Right, maxVal)

    // обновляем глобальный максимум диаметра
    *maxVal = max(*maxVal, left+right)

    // возвращаем высоту поддерева
    return max(left, right) + 1
}

func diameterOfBinaryTree(root *TreeNode) int {
    if root == nil {
        return 0
    }

    maxVal := 0
    findMax(root, &maxVal)
    return maxVal
}
```
Идея решения
- **Высота поддерева (depth)**: количество узлов от текущего до самого глубокого листа.  
- **Диаметр**: количество рёбер на самом длинном пути.  
- На каждом узле возможный диаметр = `leftDepth + rightDepth`.  
- Возвращаем из рекурсии **высоту**, а побочно обновляем глобальный максимум диаметра.
