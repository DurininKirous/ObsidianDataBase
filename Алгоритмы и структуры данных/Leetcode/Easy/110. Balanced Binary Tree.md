```go
func isBalanced(root *TreeNode) bool {
    return checkHeight(root) != -1
}

func checkHeight(node *TreeNode) int {
    if node == nil {
        return 0
    }

    left := checkHeight(node.Left)
    if left == -1 {
        return -1
    }
    right := checkHeight(node.Right)
    if right == -1 {
        return -1
    }

    if abs(left-right) > 1 {
        return -1
    }

    return max(left, right) + 1
}

func abs(x int) int {
    if x < 0 {
        return -x
    }
    return x
}

func max(a, b int) int {
    if a > b {
        return a
    }
    return b
}
```
## Идея решения
- Для каждого узла нужно знать **высоту его поддерева**.
- Проверяем разницу высот:
    - если больше `1` → дерево несбалансировано.
- Если хоть одно поддерево несбалансировано → всё дерево несбалансировано.
- Чтобы не пересчитывать высоту каждый раз, возвращаем высоту в рекурсии сразу.
    
⚡️ Приём: если нашли несбалансированное поддерево, возвращаем специальное значение (`-1`), чтобы дальше не считать.
