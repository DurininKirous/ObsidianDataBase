```go
type MyQueue struct {
    numsIn  []int
    numsOut []int
}

func Constructor() MyQueue {
    return MyQueue{numsIn: []int{}, numsOut: []int{}}
}

func (this *MyQueue) Push(x int) {
    this.numsIn = append(this.numsIn, x)
}

func (this *MyQueue) Pop() int {
    if len(this.numsOut) == 0 {
        for len(this.numsIn) != 0 {
            this.numsOut = append(this.numsOut, this.numsIn[len(this.numsIn)-1])
            this.numsIn = this.numsIn[:len(this.numsIn)-1]
        }
    }
    val := this.numsOut[len(this.numsOut)-1]
    this.numsOut = this.numsOut[:len(this.numsOut)-1]
    return val
}

func (this *MyQueue) Peek() int {
    if len(this.numsOut) == 0 {
        for len(this.numsIn) != 0 {
            this.numsOut = append(this.numsOut, this.numsIn[len(this.numsIn)-1])
            this.numsIn = this.numsIn[:len(this.numsIn)-1]
        }
    }
    return this.numsOut[len(this.numsOut)-1]
}

func (this *MyQueue) Empty() bool {
    return len(this.numsIn) == 0 && len(this.numsOut) == 0
}
```

Идея (по моему решению)
- Используем **два стека** (`numsIn`, `numsOut`):
  - `numsIn` — принимает новые элементы через `Push`.
  - `numsOut` — отдаёт элементы через `Pop` и `Peek`.
- При `Pop` и `Peek`:  
  если `numsOut` пуст, перекладываем все элементы из `numsIn` в `numsOut`, тем самым разворачивая порядок, чтобы первый вставленный элемент оказался сверху `numsOut`.
