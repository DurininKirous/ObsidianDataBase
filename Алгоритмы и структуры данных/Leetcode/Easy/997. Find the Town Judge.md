```go
func findJudge(n int, trust [][]int) int {
    score := make([]int, n+1) // score[i] = in(i) - out(i)
    for _, t := range trust {
        a, b := t[0], t[1]
        score[a]-- // a кому-то доверяет -> не судья
        score[b]++ // b получает доверие
    }
    for i := 1; i <= n; i++ {
        if score[i] == n-1 { return i }
    }
    return -1
}
```
## Идея
- Судья: ему доверяют все (`in = n-1`), он никому не доверяет (`out = 0`).
- Храним `score[i] = in - out`; ищем `score[i] == n-1`.

## Сложность
- Время: `O(n + |trust|)`
- Память: `O(n)`
