#sr-due 
import "unicode"

func isAlnum(r rune) bool {
    return unicode.IsLetter(r) || unicode.IsDigit(r)
}

func isPalindrome(s string) bool {
    runes := []rune(s)
    left, right := 0, len(runes)-1

    for left < right {
        for left < right && !isAlnum(runes[left]) {
            left++
        }
        for left < right && !isAlnum(runes[right]) {
            right--
        }
        if unicode.ToLower(runes[left]) != unicode.ToLower(runes[right]) {
            return false
        }
        left++
        right--
    }
    return true
}

### Концепция
- Используем два указателя: `left` и `right`.
- Игнорируем все не-alphanumeric символы (пробелы, знаки препинания).
- Сравниваем символы без учёта регистра (через `unicode.ToLower`).
- Если встречаем несовпадение — строка не палиндром.
- Двигаем указатели к центру, пока `left < right`.
### 📜 Алгоритм
1. Преобразовать строку в `[]rune`, чтобы корректно работать с Unicode.
2. Установить `left = 0`, `right = len(runes)-1`.
3. Пока `left < right`:
    - Сдвигать `left`, пока не найдём букву или цифру.
    - Сдвигать `right`, пока не найдём букву или цифру.
    - Сравнить символы без учёта регистра:
        - Если не равны → `false`.
    - Сдвинуть оба указателя к центру.
4. Вернуть `true`, если цикл завершился без расхождений.
