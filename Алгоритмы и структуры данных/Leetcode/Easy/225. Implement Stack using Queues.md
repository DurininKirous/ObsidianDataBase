```go
type MyStack struct {
    q1 []int
    q2 []int
}

func Constructor() MyStack {
    return MyStack{q1: []int{}, q2: []int{}}
}

func (this *MyStack) Push(x int) {
    // кладём новый элемент во вторую очередь
    this.q2 = append(this.q2, x)

    // переносим все элементы из q1 в q2
    for len(this.q1) > 0 {
        this.q2 = append(this.q2, this.q1[0])
        this.q1 = this.q1[1:]
    }

    // меняем местами
    this.q1, this.q2 = this.q2, this.q1
}

func (this *MyStack) Pop() int {
    val := this.q1[0]
    this.q1 = this.q1[1:]
    return val
}

func (this *MyStack) Top() int {
    return this.q1[0]
}

func (this *MyStack) Empty() bool {
    return len(this.q1) == 0
}

```

### `push O(n)`, `pop O(1)`
- При `push(x)` добавляем в `q2`, переносим все элементы из `q1` → `q2`.
- Меняем местами `q1` и `q2`.
- В итоге верхушка стека всегда в начале `q1`.
