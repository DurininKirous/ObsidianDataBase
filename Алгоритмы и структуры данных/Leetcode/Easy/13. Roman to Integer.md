func romanToInt(s string) int {
    trans := map[byte]int{
        'I':1, 'V':5, 'X':10,
        'L':50, 'C':100, 'D':500, 'M':1000,
    }
    sum := trans[s[len(s)-1]]
    ptr := len(s)-2
    for ptr >= 0 {
        if trans[s[ptr]] < trans[s[ptr+1]] {
            sum -= trans[s[ptr]]
        } else {
            sum += trans[s[ptr]]
        }
        ptr--
    }
    return sum
}

## Алгоритм
- Создать таблицу значений (`map[byte]int`).
- Идти справа налево.
- Если текущий символ < предыдущего → вычесть.
- Иначе → прибавить.
- Сумма в конце — ответ.
