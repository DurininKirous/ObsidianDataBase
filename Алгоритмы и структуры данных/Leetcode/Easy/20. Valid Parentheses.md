func isValid(s string) bool {
    stack := make([]rune, 0, len(s))
    set := map[rune]rune{
        '(': ')',
        '{': '}',
        '[': ']',
    }
    for _, val := range s {
        if _, exists := set[val]; exists {
            stack = append(stack, val)
        } else {
            if len(stack) == 0 || set[stack[len(stack)-1]] != val {
                return false
            }
            stack = stack[:len(stack)-1]
        }
    }
    return len(stack) == 0
}

## Алгоритм
- Использовать **стек** для отслеживания открывающих скобок.
- Когда встречается открывающая (`(`, `{`, `[`), кладём её в стек.
- Когда встречается закрывающая (`)`, `}`, `]`):
    - если стек пуст → false
    - если верх стека не соответствует → false
    - иначе снимаем верхний элемент из стека (**pop**).
- В конце стек должен быть пуст.
