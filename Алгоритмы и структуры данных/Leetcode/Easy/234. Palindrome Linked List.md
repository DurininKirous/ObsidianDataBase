```go
func isPalindrome(head *ListNode) bool {
    if head == nil || head.Next == nil {
        return true
    }

    // ищем середину
    slow, fast := head, head
    for fast != nil && fast.Next != nil {
        slow = slow.Next
        fast = fast.Next.Next
    }

    // разворачиваем вторую половину
    var prev *ListNode
    curr := slow
    for curr != nil {
        next := curr.Next
        curr.Next = prev
        prev = curr
        curr = next
    }

    // сравниваем половины
    left, right := head, prev
    for right != nil {
        if left.Val != right.Val {
            return false
        }
        left = left.Next
        right = right.Next
    }
    return true
}

```
1. Найти середину списка с помощью **fast/slow указателей**.
2. Развернуть вторую половину списка.
3. Сравнить первую половину и развернутую вторую.
4. (Опционально) восстановить список в исходное состояние.
	- Время: **O(n)**
	- Память: **O(1)**
