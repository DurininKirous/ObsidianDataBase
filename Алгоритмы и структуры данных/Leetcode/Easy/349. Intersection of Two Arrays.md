```go
func intersection(nums1 []int, nums2 []int) []int {
    s1 := make(map[int]bool)
    res := []int{}
    for _, val := range nums1 {
        s1[val] = true
    }

    for _, val := range nums2 {
        if s1[val] {
            res = append(res, val)
            s1[val] = false // предотвращает дубликаты
        }
    }
    return res
}
```
### 💡 Идея
Найти уникальные элементы, которые встречаются в обоих массивах.  
Используем **HashSet (map)** для быстрого поиска и удаления дубликатов.
