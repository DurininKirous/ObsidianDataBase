```go
func isAnagram(s, t string) bool {
    if len(s) != len(t) {
        return false
    }
    count := make(map[rune]int)
    for _, ch := range s {
        count[ch]++
    }
    for _, ch := range t {
        count[ch]--
        if count[ch] < 0 {
            return false
        }
    }
    return true
}
```
### 💡 Идея
Нужно проверить, состоят ли две строки из одинаковых букв с одинаковыми частотами.  
Самый быстрый способ — **подсчитать частоты символов**.
