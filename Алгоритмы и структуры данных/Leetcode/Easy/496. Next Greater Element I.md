```go
func nextGreaterElement(nums1 []int, nums2 []int) []int {
    indexes := make(map[int]int) // значение → next greater
    stack := []int{}
    res := make([]int, len(nums1))

    // Заполняем map по nums2
    for _, val := range nums2 {
        for len(stack) > 0 && val > stack[len(stack)-1] {
            indexes[stack[len(stack)-1]] = val
            stack = stack[:len(stack)-1]
        }
        stack = append(stack, val)
    }
    // Остатки в стеке = -1
    for _, v := range stack {
        indexes[v] = -1
    }

    // Формируем ответ по nums1
    for i, val := range nums1 {
        res[i] = indexes[val]
    }
    return res
}
```
### Идея
Используем **монотонный стек** (убывающий):
- Идём слева направо по `nums2`.
- Пока текущий элемент больше вершины стека → для вершины найден next greater → записываем в map.
- Кладём текущий элемент в стек.
- В конце для оставшихся в стеке → next greater = -1.
- Для ответа по `nums1` просто достаём значения из map.
