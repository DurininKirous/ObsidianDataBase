func numDecodings(s string) int {
    if len(s) == 0 || s[0] == '0' {
        return 0
    }

    dp := make([]int, len(s)+1)
    dp[0] = 1
    dp[1] = 1

    letters := []byte(s)
    for i := 2; i <= len(letters); i++ {
        one := letters[i-1] - '0'
        two := (letters[i-2]-'0')*10 + one

        if one != 0 {
            dp[i] += dp[i-1]
        }
        if two >= 10 && two <= 26 {
            dp[i] += dp[i-2]
        }
    }

    return dp[len(letters)]
}
