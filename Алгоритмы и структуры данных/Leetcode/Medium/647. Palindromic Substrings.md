```go
func countSubstrings(s string) int {
    count := 0
    for i, _ := range s {
        count += substr(s, i, i)
        count += substr(s, i, i+1)
    }
    return count
}

func substr(s string, l, r int) int {
    count := 0
    for l >= 0 && r < len(s) && s[l] == s[r] {
        l--
        r++
        count++
    }
    return count
}
```
**Идея:** перебираем все центры палиндромов (нечётные `i,i` и чётные `i,i+1`) и расширяемся влево/вправо, считая совпадения.
