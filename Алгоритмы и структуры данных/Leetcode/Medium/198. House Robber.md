```go
func rob(nums []int) int {
    dp := []int{nums[0]}
    maxN := nums[0]

    if len(nums) > 1 {
        dp = append(dp, max(maxN, nums[1]))
    }

    for i := 2; i < len(nums); i++ {
        maxN = max(maxN, dp[i-2])
        dp = append(dp, nums[i] + maxN)
    }

    res := 0
    if len(dp)-2 >= 0 {
        res = max(dp[len(dp)-1], dp[len(dp)-2])
    } else {
        res = dp[len(dp)-1]
    }

    return res
}
```
### Основная идея
- Храним максимум по предыдущим “допустимым” домам (`maxN`).
- На каждом шаге обновляем `maxN` как `max(maxN, dp[i-2])`.
- Таким образом, не нужно пересчитывать максимум каждый раз.
