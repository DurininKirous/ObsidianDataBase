```python
class WordDictionary:
    def __init__(self):
        self.children = [None for _ in range(26)]
        self.isEnd = False

    def addWord(self, word: str) -> None:
        curr = self
        for ch in word:
            key = ord(ch) - ord('a')
            if curr.children[key] is None:
                curr.children[key] = WordDictionary()
            curr = curr.children[key]
        curr.isEnd = True

    def search(self, word: str) -> bool:
        def dfs(i, node):
            if node is None:
                return False
            if i == len(word):
                return node.isEnd

            ch = word[i]
            if ch == ".":
                for child in node.children:
                    if child and dfs(i + 1, child):
                        return True
                return False
            else:
                key = ord(ch) - ord('a')
                return dfs(i + 1, node.children[key])

        return dfs(0, self)
```
**Идея**  
Префиксное дерево (trie). `addWord` идёт по символам и создаёт узлы.  
`search` — DFS по trie; если символ `'.'`, перебираем **всех существующих** детей текущего узла с ранним выходом.
