```go
func maxAreaOfIsland(grid [][]int) int {
    maxN := 0
    for i := 0; i < len(grid); i++ {
        for j := 0; j < len(grid[i]); j++ {
            if grid[i][j] == 1 {
                res := Fill(&grid, i, j)
                maxN = max(maxN, res)
            }
        }
    }
    return maxN
}

func Fill(grid *[][]int, i, j int) int {
    if i < 0 || i >= len(*grid) || j < 0 || j >= len((*grid)[0]) || (*grid)[i][j] == 0 {
        return 0
    }

    (*grid)[i][j] = 0 // затопили клетку

    return 1 +
        Fill(grid, i+1, j) +
        Fill(grid, i-1, j) +
        Fill(grid, i, j+1) +
        Fill(grid, i, j-1)
}
```
## 💡 Идея
Нужно найти площадь (количество клеток) самого большого острова в матрице.
- `1` = суша
- `0` = вода
- **Остров** = связная компонента из единиц, соединённых по 4 направлениям (вверх, вниз, влево, вправо).

### Алгоритм (DFS flood fill):
1. Инициализируем `maxArea = 0`.
2. Проходим по всем клеткам матрицы.
3. Если встречаем сушу (`1`) → запускаем DFS (`Fill`) и считаем размер острова.
4. Сравниваем с текущим максимумом и обновляем.
5. Возвращаем `maxArea`.
