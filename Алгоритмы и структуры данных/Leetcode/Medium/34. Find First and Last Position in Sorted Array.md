```python
import bisect

def searchRange(nums, target):
    l = bisect.bisect_left(nums, target)
    r = bisect.bisect_right(nums, target)
    if l != len(nums) and nums[l] == target:
        return [l, r-1]
    return [-1, -1]

# Без bisect:
def searchRange(nums, target):
    def findFirst(nums, target):
        left, right = 0, len(nums) - 1
        result = -1
        while left <= right:
            mid = (left + right) // 2
            if nums[mid] == target:
                result = mid
                right = mid - 1  # ищем левее
            elif nums[mid] < target:
                left = mid + 1
            else:
                right = mid - 1
        return result
    
    def findLast(nums, target):
        left, right = 0, len(nums) - 1
        result = -1
        while left <= right:
            mid = (left + right) // 2
            if nums[mid] == target:
                result = mid
                left = mid + 1  # ищем правее
            elif nums[mid] < target:
                left = mid + 1
            else:
                right = mid - 1
        return result
    
    first = findFirst(nums, target)
    if first == -1:
        return [-1, -1]
    return [first, findLast(nums, target)]
```
**Идея**: bisect_left → первое вхождение, bisect_right → после последнего. Или два binary search: leftmost (при найденном идем влево) и rightmost (при найденном идем вправо). O(log n) time.
