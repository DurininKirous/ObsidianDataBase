```go
func rootsSum(root *TreeNode, targetSum int, curr []int, res *[][]int) {
    if root == nil {
        return
    }

    curr = append(curr, root.Val)

    // если дошли до листа и сумма совпала → копируем путь в результат
    if root.Left == nil && root.Right == nil && targetSum == root.Val {
        pathCopy := append([]int{}, curr...)
        *res = append(*res, pathCopy)
        return
    }

    rootsSum(root.Left, targetSum-root.Val, curr, res)
    rootsSum(root.Right, targetSum-root.Val, curr, res)
}

func pathSum(root *TreeNode, targetSum int) [][]int {
    res := [][]int{}
    rootsSum(root, targetSum, []int{}, &res)
    return res
}

```
## Как работает
1. `rootsSum` идёт DFS по дереву, поддерживая текущий путь (`curr`).
2. На каждом шаге добавляем `root.Val` в `curr`.
3. Если узел — **лист** и сумма совпала (`targetSum == root.Val`) → сохраняем копию пути в результат `res`.
    - Используем `append([]int{}, curr...)`, чтобы создать **копию** пути (иначе слайс изменится при возврате).
4. Иначе рекурсивно спускаемся:
    - Влево: уменьшаем сумму на `root.Val`.
    - Вправо: уменьшаем сумму на `root.Val`.
5. `pathSum` создаёт пустой результат и запускает рекурсию.
