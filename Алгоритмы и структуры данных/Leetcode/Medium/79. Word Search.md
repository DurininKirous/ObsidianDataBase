```go
func exist(board [][]byte, word string) bool {
    for i := 0; i < len(board); i++ {
        for j := 0; j < len(board[0]); j++ {
            if board[i][j] == word[0] && dfs(board, word, i, j, 0){
                return true
            }
        }
    }
    return false
}

func dfs(board [][]byte, word string, i, j, k int) bool {
    if k == len(word) {
        return true
    }

    if i < 0 || i >= len(board) || j < 0 || j >= len(board[0]) || board[i][j] != word[k] {
        return false
    }
    ch := board[i][j]
    board[i][j] = '#'
    res := dfs(board, word, i - 1, j, k+1) || dfs(board, word, i + 1, j, k+1) || dfs(board, word, i, j - 1, k+1) || dfs(board, word, i, j + 1, k+1)
    board[i][j] = ch
    return res
}
```
**Идея:**
- Стартуем из каждой клетки, где совпадает первая буква.
- Рекурсивный DFS пытается собрать слово, двигаясь по 4 направлениям.
- Чтобы не переиспользовать клетку в одном пути, временно помечаем её `'#'` и **обязательно восстанавливаем** после рекурсии (backtracking).
