import "sort"

func merge(intervals [][]int) [][]int {
    sort.Slice(intervals, func(i, j int) bool {
        return intervals[i][0] < intervals[j][0] ||
            (intervals[i][0] == intervals[j][0] && intervals[i][1] < intervals[j][1])
    })

    left, right := intervals[0][0], intervals[0][1]
    res := [][]int{}

    for i := 1; i < len(intervals); i++ {
        if intervals[i][0] <= right {
            if intervals[i][1] > right {
                right = intervals[i][1]
            }
        } else {
            res = append(res, []int{left, right})
            left, right = intervals[i][0], intervals[i][1]
        }
    }
    res = append(res, []int{left, right})
    return res
}

## Алгоритм
1. Отсортировать интервалы по `start`, при равенстве — по `end`.
2. Взять первый интервал как текущий `[left, right]`.
3. Для каждого следующего интервала:
    - Если `start <= right` → расширяем интервал: `right = max(right, end)`.
    - Иначе → добавляем `[left, right]` в результат и обновляем текущий интервал.
4. Добавить последний интервал в результат.
