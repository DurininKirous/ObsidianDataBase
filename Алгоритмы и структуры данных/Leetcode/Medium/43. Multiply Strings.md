```python
class Solution:
    def multiply(self, num1: str, num2: str) -> str:
        if num1 == "0" or num2 == "0":
            return "0"

        n, m = len(num1), len(num2)
        res = [0] * (n + m)

        for i in range(n - 1, -1, -1):
            d1 = ord(num1[i]) - ord('0')
            for j in range(m - 1, -1, -1):
                d2 = ord(num2[j]) - ord('0')
                mul = d1 * d2
                p2 = i + j + 1
                p1 = i + j
                s = mul + res[p2]
                res[p2] = s % 10
                res[p1] += s // 10

        i = 0
        while i < len(res) and res[i] == 0:
            i += 1
        return "".join(str(d) for d in res[i:]) or "0"
```
**Идея:**  
симулируем умножение в столбик. Делаем массив длиной `n + m`, идём с конца обеих строк, перемножаем цифры, складываем в массив с учётом переноса.
