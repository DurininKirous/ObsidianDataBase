```go
func makeVisitArray(heights [][]int) [][]bool {
    rows := len(heights)
    col := len(heights[0])
    visit := make([][]bool, rows)
    for  i := 0; i < rows; i++ {
        visit[i] = make([]bool, col)
    }
    return visit
}

func dfs(r, c int, heights [][]int, visited [][]bool, prev int) {
    if r < 0 || c < 0 || r >= len(heights) || c >= len(heights[0]) {
        return 
    } 
    if  visited[r][c] || heights[r][c] < prev {
        return 
    }
    visited[r][c] = true
    dfs(r-1, c, heights, visited, heights[r][c])
    dfs(r+1, c, heights, visited, heights[r][c])
    dfs(r, c-1, heights, visited, heights[r][c])
    dfs(r, c+1, heights, visited, heights[r][c])
}

func runDFSFromOcean(heights [][]int, visited [][]bool, pacific bool) {
    rows := len(heights)
    col := len(heights[0])
    if pacific {
        for i := 0; i < rows; i++ {
            dfs(i, 0, heights, visited, heights[i][0])
        }
        for i := 0; i < col; i++ {
            dfs(0, i, heights, visited, heights[0][i])
        }
    } else {
        for i := 0; i < rows; i++ {
            dfs(i, len(heights[0])-1, heights, visited, heights[i][len(heights[0])-1])
        }
        for i := 0; i < col; i++ {
            dfs(len(heights)-1, i, heights, visited, heights[len(heights)-1][i])
        }
    }
}

func collectResult(pacific, atlantic [][]bool) [][]int {
    rows, cols := len(pacific), len(pacific[0])
    res := [][]int{}
    for r := 0; r < rows; r++ {
        for c := 0; c < cols; c++ {
            if pacific[r][c] && atlantic[r][c] {
                res = append(res, []int{r, c})
            }
        }
    }
    return res
}

func pacificAtlantic(heights [][]int) [][]int {
    pacific := makeVisitArray(heights)
    atlantic := makeVisitArray(heights)

    runDFSFromOcean(heights, pacific, true)   
    runDFSFromOcean(heights, atlantic, false)

    return collectResult(pacific, atlantic)
}
```

## Идея
Дана матрица высот `heights`.
- Вода может течь **вниз или вбок** (в соседние клетки с **равной или меньшей высотой**).
- Тихий океан (Pacific) касается **верхней и левой границы**, Атлантический (Atlantic) — **нижней и правой границы**.
- Нужно найти все клетки, из которых вода может достичь **обоих океанов**.

### Ключевая мысль
Вместо запуска воды **из каждой клетки** → запускаем DFS/BFS **от океанов внутрь**.
- Отмечаем клетки, достижимые из Pacific → массив `pacific`.
- Отмечаем клетки, достижимые из Atlantic → массив `atlantic`.
- Ответ = пересечение этих массивов.
