```go
func nextGreaterElements(nums []int) []int {
    stack := []int{}
    res := make([]int, len(nums))
    for i := range res {
        res[i] = -1
    }
    for i := 0; i < 2*len(nums); i++ {
        for len(stack) > 0 && nums[i%len(nums)] > nums[stack[len(stack)-1]] {
            res[stack[len(stack)-1]] = nums[i%len(nums)]
            stack = stack[:len(stack)-1]
        }
        if i < len(nums) {
            stack = append(stack, i%len(nums))
        }
    }
    return res
}
```
### Концепция решения
- Используем **монотонный стек** с индексами.
- Делаем **2 прохода по массиву** (эмулируем закольцованность).
- Для каждого элемента:
    1. Пока текущий элемент больше вершины стека → он является next greater для вершины → записываем в `res`.
    2. В первой половине прохода кладём индекс в стек.
- `res` изначально заполняем `-1`.
