```python
class Solution:  
def minReorder(self, n: int, connections: List[List[int]]) -> int:  
adj = [[] for _ in range(n)]  
  
for a, b in connections:  
adj[a].append((b, True))  
adj[b].append((a, False))  
  
visited = [False] * n  
  
def dfs(node):  
visited[node] = True  
changes = 0  
  
for neighbor, is_original in adj[node]:  
if not visited[neighbor]:  
if is_original:  
changes += 1  
changes += dfs(neighbor)  
  
return changes  
  
return dfs(0)
```
