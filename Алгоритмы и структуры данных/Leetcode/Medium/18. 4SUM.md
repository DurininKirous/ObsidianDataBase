#sr-due 
import "sort"

func fourSum(nums []int, target int) [][]int {
    sort.Ints(nums)
    res := [][]int{}

    for i := 0; i < len(nums)-3; i++ {
        if i > 0 && nums[i] == nums[i-1] {
            continue
        }
        for j := i + 1; j < len(nums)-2; j++ {
            if j > i+1 && nums[j] == nums[j-1] {
                continue
            }
            left, right := j+1, len(nums)-1
            for left < right {
                sum := nums[i] + nums[j] + nums[left] + nums[right]
                if sum < target {
                    left++
                } else if sum > target {
                    right--
                } else {
                    res = append(res, []int{nums[i], nums[j], nums[left], nums[right]})
                    for left < right && nums[left] == nums[left+1] {
                        left++
                    }
                    for left < right && nums[right] == nums[right-1] {
                        right--
                    }
                    left++
                    right--
                }
            }
        }
    }
    return res
}

## Концепция
- Найти все уникальные quadruplets `(a,b,c,d)` с суммой `target`.
- Используем **сортировку** + **два вложенных цикла** + **two pointers**.

## Алгоритм
1. `sort.Ints(nums)`
2. Цикл по `i` (`0..n-4`), пропускаем дубликаты.
3. Цикл по `j` (`i+1..n-3`), пропускаем дубликаты.
4. `left=j+1`, `right=n-1`
5. Вычисляем `sum`
    - если `< target` → `left++`
    - если `> target` → `right--`
    - если `== target` → сохраняем quadruplet, пропускаем дубликаты для left/right, двигаем оба указателя.
