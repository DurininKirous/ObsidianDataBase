type MinStack struct {
    items []int
    mins  []int
}

func Constructor() MinStack {
    return MinStack{items: []int{}, mins: []int{}}
}

func (this *MinStack) Push(val int) {
    this.items = append(this.items, val)
    this.mins = append(this.mins, val)
    if len(this.mins) > 1 && val > this.mins[len(this.mins)-2] {
        this.mins[len(this.mins)-1] = this.mins[len(this.mins)-2]
    }
}

func (this *MinStack) Pop() {
    this.items = this.items[:len(this.items)-1]
    this.mins = this.mins[:len(this.mins)-1]
}

func (this *MinStack) Top() int {
    return this.items[len(this.items)-1]
}

func (this *MinStack) GetMin() int {
    return this.mins[len(this.mins)-1]
}

### Алгоритм (двойной стек)
1. Храним два стека:
    - `items` — обычные значения;
    - `mins` — текущий минимум на каждый момент времени.
2. При `Push(val)` кладём значение в `items`.  
    В `mins` кладём либо `val`, либо предыдущий минимум (что меньше).
3. При `Pop()` убираем элементы из обоих стэков.
4. При `Top()` возвращаем последний элемент из `items`.
5. При `GetMin()` возвращаем последний элемент из `mins`.### Алгоритм (двойной стек)
