```
func addTwoNumbers(l1 *ListNode, l2 *ListNode) *ListNode {
    dummy := &ListNode{}
    curr := dummy
    carry := 0

    for l1 != nil || l2 != nil || carry > 0 {
        sum := carry
        if l1 != nil {
            sum += l1.Val
            l1 = l1.Next
        }
        if l2 != nil {
            sum += l2.Val
            l2 = l2.Next
        }

        curr.Next = &ListNode{Val: sum % 10}
        curr = curr.Next
        carry = sum / 10
    }
    return dummy.Next
}
```
## Концепция
- Используем **dummy head** для построения нового списка.
- Идём по обоим спискам одновременно.
- На каждой итерации считаем сумму: `val1 + val2 + carry`.
- Создаём новый узел со значением `sum % 10`.
- Перенос (`carry = sum / 10`) сохраняем для следующего шага.
- Если в конце `carry > 0` → добавляем ещё один узел.
