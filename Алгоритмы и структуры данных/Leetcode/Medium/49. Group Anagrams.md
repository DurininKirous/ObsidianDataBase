import "sort"

func sortString(str string) string {
    strB := []byte(str)
    sort.Slice(strB, func(i, j int) bool { return strB[i] < strB[j] })
    return string(strB)
}

func groupAnagrams(strs []string) [][]string {
    words := make(map[string][]string)
    for _, val := range strs {
        key := sortString(val)
        words[key] = append(words[key], val)
    }
    res := [][]string{}
    for _, val := range words {
        res = append(res, val)
    }
    return res
}

## Алгоритм (подход через сортировку)
1. Для каждой строки получить **ключ**: отсортированную версию строки (`"eat" → "aet"`).
2. Использовать `map[string][]string`, где ключ = отсортированная строка, значение = список анаграмм.
3. Пройти по всем словам и добавить их в мапу.
4. Собрать значения мапы в `[][]string`.
