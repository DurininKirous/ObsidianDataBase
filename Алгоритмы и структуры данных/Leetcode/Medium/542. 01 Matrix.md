```go
func updateMatrix(mat [][]int) [][]int {
    rows, cols := len(mat), len(mat[0])
    dist := make([][]int, rows)
    for i := range dist {
        dist[i] = make([]int, cols)
    }

    const INF = 1 << 30
    q := make([][2]int, 0, rows*cols)

    // инициализация: нули = 0, единицы = INF
    for r := 0; r < rows; r++ {
        for c := 0; c < cols; c++ {
            if mat[r][c] == 0 {
                dist[r][c] = 0
                q = append(q, [2]int{r, c})
            } else {
                dist[r][c] = INF
            }
        }
    }

    dirs := [][2]int{{1,0},{-1,0},{0,1},{0,-1}}

    // BFS
    for head := 0; head < len(q); head++ {
        r, c := q[head][0], q[head][1]
        for _, d := range dirs {
            nr, nc := r+d[0], c+d[1]
            if nr >= 0 && nr < rows && nc >= 0 && nc < cols &&
               dist[nr][nc] > dist[r][c]+1 {
                dist[nr][nc] = dist[r][c]+1
                q = append(q, [2]int{nr, nc})
            }
        }
    }
    return dist
}
```
## Идея
Дана бинарная матрица `mat` (`0` и `1`).  
Нужно для каждой клетки вернуть расстояние до ближайшего `0`.  
Расстояние = количество шагов по соседним клеткам (вверх/вниз/влево/вправо).
### Ключевая мысль
- Если идти от каждой `1` искать `0` — будет слишком дорого (`O((mn)²)`).
- Оптимально идти **от всех нулей сразу** (multi-source BFS).
- Задаём все `0` в очередь с расстоянием `0`.
- Остальные (`1`) инициализируем как ∞.
- Запускаем BFS волнами: каждый сосед получает значение `dist[curr]+1`, если оно меньше текущего.
