```go
import "slices"

func levelOrderBottom(root *TreeNode) [][]int {
    type NodeLevel struct {
        Node  *TreeNode
        Level int
    }

    res := [][]int{}
    if root == nil {
        return res
    }

    queue := []*NodeLevel{{Node: root, Level: 0}}

    for len(queue) > 0 {
        node := queue[0]
        queue = queue[1:]

        if len(res) <= node.Level {
            res = append(res, []int{})
        }

        res[node.Level] = append(res[node.Level], node.Node.Val)

        if node.Node.Left != nil {
            queue = append(queue, &NodeLevel{Node: node.Node.Left, Level: node.Level + 1})
        }
        if node.Node.Right != nil {
            queue = append(queue, &NodeLevel{Node: node.Node.Right, Level: node.Level + 1})
        }
    }

    slices.Reverse(res)
    return res
}
```

Идея решения
- Используем **BFS с очередью**.  
- Храним в очереди узлы вместе с их уровнем.  
- Для каждого узла добавляем его значение в `res[level]`.  
- После обхода делаем `slices.Reverse(res)` (Go 1.21+), чтобы перевернуть уровни.
