func detectCycle(head *ListNode) *ListNode {
    slow := head
    fast := head
    for fast != nil && fast.Next != nil {
        slow = slow.Next
        fast = fast.Next.Next
        if slow == fast {
            break
        }
    }

    if fast == nil || fast.Next == nil {
        return nil
    }

    fast = head
    for fast != slow {
        fast = fast.Next
        slow = slow.Next
    }
    return slow
}
## Алгоритм (Floyd’s Cycle Detection, расширение)
1. Два указателя:
    - `slow` двигается на 1 шаг,
    - `fast` двигается на 2 шага.
2. Если `slow == fast`, найден цикл.  
    Если `fast == nil` или `fast.Next == nil` → цикла нет.
3. Перезапустить один указатель с головы (`fast = head`).
4. Двигать `fast` и `slow` по одному шагу.
5. Точка их встречи = начало цикла.
