```go
func findPaths(root *TreeNode, curr int, sum *int) {
    if root == nil {
        return
    }

    curr = curr*10 + root.Val

    if root.Left == nil && root.Right == nil {
        *sum += curr
        return
    }
    findPaths(root.Left, curr, sum)
    findPaths(root.Right, curr, sum)
}

func sumNumbers(root *TreeNode) int {
    if root == nil {
        return 0
    }
    sum := 0
    findPaths(root, 0, &sum)
    return sum
}
```
## Идея
- Идём DFS от корня к листьям.
- На каждом шаге строим число:
	curr = curr * 10 + root.Val
- Если дошли до листа → добавляем число к сумме.
