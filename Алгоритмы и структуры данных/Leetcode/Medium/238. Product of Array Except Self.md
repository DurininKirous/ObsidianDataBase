func productExceptSelf(nums []int) []int {
    n := len(nums)
    res := make([]int, n)

    // префиксы
    res[0] = 1
    for i := 1; i < n; i++ {
        res[i] = res[i-1] * nums[i-1]
    }

    // суффиксы
    suffix := 1
    for i := n-1; i >= 0; i-- {
        res[i] *= suffix
        suffix *= nums[i]
    }

    return res
}

## Алгоритм (prefix/suffix products)
1. Считаем произведение **всех элементов слева** от `i` → кладём в `res[i]`.
2. Второй проход справа: умножаем `res[i]` на произведение всех элементов справа от `i`.
3. Используем одну переменную `suffix` для накопления
