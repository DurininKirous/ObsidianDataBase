```go
/**
 * Definition for a binary tree node.
 * type TreeNode struct {
 *     Val   int
 *     Left  *TreeNode
 *     Right *TreeNode
 * }
 */

func buildTree(preorder []int, inorder []int) *TreeNode {
    if len(preorder) == 0 || len(inorder) == 0 {
        return nil
    }

    // map для быстрого поиска индекса в inorder
    indexMap := make(map[int]int)
    for i, v := range inorder {
        indexMap[v] = i
    }

    return build(preorder, 0, len(preorder)-1, inorder, 0, len(inorder)-1, indexMap)
}

func build(preorder []int, preL, preR int, inorder []int, inL, inR int, indexMap map[int]int) *TreeNode {
    if preL > preR || inL > inR {
        return nil
    }

    rootVal := preorder[preL]
    root := &TreeNode{Val: rootVal}

    inRoot := indexMap[rootVal]          // индекс корня в inorder
    leftSize := inRoot - inL             // размер левого поддерева

    root.Left = build(preorder, preL+1, preL+leftSize, inorder, inL, inRoot-1, indexMap)
    root.Right = build(preorder, preL+leftSize+1, preR, inorder, inRoot+1, inR, indexMap)

    return root
}
```

Идея решения
1. В `preorder` первым всегда идёт **корень** поддерева.
2. В `inorder` все элементы **слева от корня** принадлежат левому поддереву, а все **справа** — правому.
3. Зная индекс корня в `inorder`, вычисляем размер левого поддерева.
4. Рекурсивно строим:
   - левое поддерево из соответствующих частей `preorder` и `inorder`,
   - правое поддерево аналогично.
5. Чтобы быстро находить позицию корня в `inorder`, используем `map[value] = index`.
