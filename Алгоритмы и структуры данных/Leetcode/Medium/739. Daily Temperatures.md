```go
func dailyTemperatures(temperatures []int) []int {
    stack := []int{}
    res := make([]int, len(temperatures))

    for i, val := range temperatures {
        for len(stack) > 0 && val > temperatures[stack[len(stack)-1]] {
            prev := stack[len(stack)-1]
            stack = stack[:len(stack)-1]
            res[prev] = i - prev
        }
        stack = append(stack, i)
    }

    return res
}
```
### Концепция решения
- Используем **монотонный стек** (убывающий по температурам).
- Стек хранит **индексы** температур.
- Когда встречаем температуру выше, чем на вершине стека:
    - достаём индекс из стека,
    - считаем разницу `i - prev` и пишем в `res[prev]`.
- В конце все оставшиеся индексы автоматически имеют `0`.
