func detectCycle(head *ListNode) *ListNode {
    seen := make(map[*ListNode]struct{})
    for head != nil {
        if _, ok := seen[head]; ok {
            return head
        }
        seen[head] = struct{}{}
        head = head.Next
    }
    return nil
}

### Алгоритм (HashMap-подход)
1. Создать `map[*ListNode]struct{}`.
2. Идти по списку, на каждом шаге:
    - Если `head` уже в `map` → вернуть `head`.
    - Иначе добавить `head` в `map`.
3. Если дошли до `nil`, вернуть `nil`.
