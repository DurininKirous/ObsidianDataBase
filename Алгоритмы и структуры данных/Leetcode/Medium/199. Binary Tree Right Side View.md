```go
func rightSideView(root *TreeNode) []int {
    if root == nil {
        return []int{}
    }

    queue := []*TreeNode{root}
    res := []int{root.Val}

    for len(queue) > 0 {
        size := len(queue)
        for i := 0; i < size; i++ {
            node := queue[0]
            queue = queue[1:]

            if node.Left != nil {
                queue = append(queue, node.Left)
            } 
            if node.Right != nil {
                queue = append(queue, node.Right)
            }
        }
        // добавляем последний элемент уровня, если есть
        if len(queue) > 0 {
            res = append(res, queue[len(queue)-1].Val)
        }
    }
    return res
}
```
💡 Идея решения
- Используем **BFS (уровневый обход)**.  
- На каждом уровне берём **последний узел** — это и будет вид с правой стороны.  
- Для этого:
  - берём размер текущего уровня `size := len(queue)`,  
  - обрабатываем `size` элементов, добавляя их детей в очередь,  
  - после цикла смотрим на последний элемент очереди, если он есть.
