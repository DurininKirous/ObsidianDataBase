import "sort"

func absInt(x int) int {
    if x < 0 {
        return -x
    }
    return x
}

func threeSumClosest(nums []int, target int) int {
    sort.Ints(nums)
    bestSum := nums[0] + nums[1] + nums[2]

    for i := 0; i < len(nums)-2; i++ {
        left, right := i+1, len(nums)-1
        for left < right {
            sum := nums[i] + nums[left] + nums[right]
            if absInt(target-sum) < absInt(target-bestSum) {
                bestSum = sum
            }
            if sum < target {
                left++
            } else if sum > target {
                right--
            } else {
                return target // оптимизация
            }
        }
    }
    return bestSum
}

## Концепция
- Найти три числа, сумма которых ближе всего к `target`.
- Массив сначала сортируем → можно использовать **два указателя**.
- Минимизируем `|sum - target|` и сохраняем лучший результат.

## Алгоритм
1. `sort.Ints(nums)` — сортируем массив.
2. Итерируем `i` (первое число).
3. Два указателя `left=i+1`, `right=len(nums)-1`.
4. Считаем `sum = nums[i]+nums[left]+nums[right]`.
    - Если `|sum-target|` меньше, чем у bestSum → обновляем.
    - Если `sum < target` → `left++`.
    - Если `sum > target` → `right--`.
    - Если `sum == target` → возвращаем target (лучше не будет).
5. Возвращаем `bestSum`.
