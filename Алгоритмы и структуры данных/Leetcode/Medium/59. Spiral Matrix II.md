```python
class Solution:
    def generateMatrix(self, n: int) -> List[List[int]]:
        curr = 0
        total = n * n
        res = [[0] * n for _ in range(n)]

        wall_up = 0
        wall_down = n
        wall_left = 0
        wall_right = n

        i = j = 0

        while curr < total:
            # вправо
            while curr < total and j < wall_right:
                curr += 1
                res[i][j] = curr
                j += 1
            wall_up += 1
            j -= 1
            i += 1

            # вниз
            while curr < total and i < wall_down:
                curr += 1
                res[i][j] = curr
                i += 1
            wall_right -= 1
            j -= 1
            i -= 1

            # влево
            while curr < total and j >= wall_left:
                curr += 1
                res[i][j] = curr
                j -= 1
            wall_down -= 1
            j += 1
            i -= 1

            # вверх
            while curr < total and i >= wall_up:
                curr += 1
                res[i][j] = curr
                i -= 1
            wall_left += 1
            j += 1
            i += 1

        return res

```
**Идея:**  
двигаемся слоями по часовой стрелке, держим “стенки”:
- `wall_up`, `wall_down`, `wall_left`, `wall_right` — текущие границы,
- после прохода вправо поджимаем верх, после прохода вниз — правую, потом низ, потом левую.
