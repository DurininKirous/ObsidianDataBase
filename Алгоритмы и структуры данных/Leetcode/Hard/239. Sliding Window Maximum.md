```go
func maxSlidingWindow(nums []int, k int) []int {
    maxNums := []int{}
    deque := []int{} // храним индексы

    if len(nums) < k {
        return maxNums
    }

    for i := 0; i < len(nums); i++ {
        // 1. Убираем индекс, вышедший за окно
        if len(deque) > 0 && deque[0] <= i-k {
            deque = deque[1:]
        }
        // 2. Убираем все меньшие элементы с хвоста
        for len(deque) > 0 && nums[i] > nums[deque[len(deque)-1]] {
            deque = deque[:len(deque)-1]
        }
        // 3. Добавляем текущий индекс
        deque = append(deque, i)

        // 4. Если окно полно — фиксируем максимум
        if i >= k-1 {
            maxNums = append(maxNums, nums[deque[0]])
        }
    }
    return maxNums
}

```
### Концепция
- **Наивный способ:** каждый раз искать максимум по окну → O(n*k).
- **Оптимальный способ (deque):**
    - храним индексы элементов в порядке убывания значений;
    - убираем из головы, если индекс вышел за окно;
    - убираем из хвоста все меньшие элементы перед вставкой;
    - максимум всегда в `nums[deque[0]]`.
Итог: **O(n)** по времени и **O(k)** по памяти.
### 🔹 Алгоритм
1. Для каждого индекса `i`:
    - удалить `deque[0]`, если он вне окна (`<= i-k`);
    - удалить из хвоста индексы элементов меньше текущего;
    - добавить `i` в хвост;
    - если окно готово (`i >= k-1`) → добавить `nums[deque[0]]` в результат.
