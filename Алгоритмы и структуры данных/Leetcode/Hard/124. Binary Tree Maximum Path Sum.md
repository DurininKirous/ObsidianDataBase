```go
/**
 * Definition for a binary tree node.
 * type TreeNode struct {
 *     Val   int
 *     Left  *TreeNode
 *     Right *TreeNode
 * }
 */

func findMax(root *TreeNode, maxVal *int) int {
    if root == nil {
        return 0
    }

    // считаем вклады от детей
    left := findMax(root.Left, maxVal)
    right := findMax(root.Right, maxVal)

    // отрицательные вклады игнорируем
    if left < 0 {
        left = 0
    }
    if right < 0 {
        right = 0
    }

    // путь, замкнутый на текущем узле
    pathSum := root.Val + left + right
    if pathSum > *maxVal {
        *maxVal = pathSum
    }

    // возвращаем максимум пути наверх (узел + одна ветка)
    if left > right {
        return root.Val + left
    }
    return root.Val + right
}

func maxPathSum(root *TreeNode) int {
    maxVal := -1 << 31 // минимальное int (лучше math.MinInt)
    findMax(root, &maxVal)
    return maxVal
}
```

Идея решения
- На каждом узле возможны два варианта:
  1. **Вклад наверх** — путь, который продолжается через родителя.  
     Для этого можно взять только одну ветку:  
     ```
     gain(node) = node.Val + max(0, gain(left), gain(right))
     ```
  2. **Замкнутый путь через текущий узел** — берём обе ветки + сам узел:  
     ```
     through = node.Val + max(0, gain(left)) + max(0, gain(right))
     ```
- Глобальный максимум обновляем через `through`.  
- В рекурсию возвращаем только `gain(node)`, чтобы родитель мог продолжить путь.
